# Platform notes:
# - Windows/Linux (x86_64): plain `tensorflow` (CPU on native Windows; GPU via WSL2).
# - macOS Intel: plain `tensorflow`.
# - macOS Apple Silicon (arm64): `tensorflow-macos` + `tensorflow-metal` (GPU).
# - OpenCV is capped at <5 because 5.x removed CascadeClassifier/haarcascades
#   used by src/detect.py.

tensorflow-macos>=2.16; sys_platform == "darwin" and platform_machine == "arm64"
tensorflow-metal; sys_platform == "darwin" and platform_machine == "arm64"
keras>=3.0

# (run_live_verification / notebook capture cells) on a Mac with display,
# install `opencv-python` instead of `opencv-python-headless`.
"opencv-python-headless>=4.10,<5"
numpy>=1.24
matplotlib>=3.7
